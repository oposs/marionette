        let conn = manager.get_connection();

        )

        let conn = manager.get_connection();

            "ALTER TABLE contact DROP COLUMN contact_opt_in",
        )
        .await?;

            "ALTER TABLE contact DROP COLUMN contact_notes",
        )
        .await?;

        )
        .await?;
