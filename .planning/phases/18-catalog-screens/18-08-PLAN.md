```rust
Button::new("")                           // no label → isIconOnly
    .id(format!("catalog-typo-icon-{name}"))
    .variant("outline")
    .size("icon")
    .icon(name)                           // kebab-name from registry
    .aria_label(format!("{name} icon"))
    // No action — icon cells are display-only, NOT interactive. See UI-SPEC
    // §CAT-05 line 592: "Cells are NOT interactive … the Button is rendered as
    // a non-action affordance". If a non-action Button produces a click target
    // the UAT flags it — but it's display-only by contract.
    .build()
```

UI-SPEC §CAT-05 line 592 allows "planner discretion" on display-only Button vs Container with icon directly. CHOICE: we use Button(size=icon) with NO action; the aria-label keeps a11y intact and the visual matches the shadcn-icon-outline look.

    //! CAT-05 — Typography & Tokens catalog screen.
    //! Three Cards stacked: Type scale, Lucide icon catalog (14 icons locked),
    //! OKLCH semantic tokens (27 swatches: 26 colour tokens + 1 radius).
    //! Pure static — no bind, no action, no interactive state.

    #![allow(clippy::too_many_lines)]

    use marionette::builders::{Button, Container, Heading, Text};
    use marionette::gallery::Node;

    const OUTER_CLASS: &str = "flex flex-col gap-6 p-6";
    const CARD_CLASS: &str =
        "rounded-lg border bg-card text-card-foreground shadow-sm p-6 flex flex-col gap-4";
    const ICON_GRID_CLASS: &str = "grid grid-cols-4 sm:grid-cols-6 lg:grid-cols-8 gap-3";
    const SWATCH_GRID_CLASS: &str = "grid grid-cols-3 sm:grid-cols-4 lg:grid-cols-6 gap-3";
    const ICON_CELL_CLASS: &str = "flex flex-col items-center gap-1 p-2 rounded border";
    const SWATCH_CELL_CLASS: &str = "flex flex-col gap-2";

    /// Icons from UI-SPEC §CAT-05 line 594 — the 14 names registered in
    /// frontend/src/lib/registry/icons.ts (kebab-case).
    const ICONS: [&str; 14] = [
        "plus", "chevron-up", "chevron-down", "alert-circle", "x", "menu",
        "arrow-left", "search", "filter", "pencil", "trash", "check", "loader",
        "circle-help",
    ];

    /// OKLCH tokens from UI-SPEC §CAT-05 lines 601-630 — 26 colour tokens in
    /// :root declaration order.
    const COLOUR_TOKENS: [&str; 26] = [
        "background", "foreground", "card", "card-foreground", "popover",
        "popover-foreground", "primary", "primary-foreground", "secondary",
        "secondary-foreground", "muted", "muted-foreground", "accent",
        "accent-foreground", "destructive", "border", "input", "ring",
        "sidebar", "sidebar-foreground", "sidebar-primary",
        "sidebar-primary-foreground", "sidebar-accent", "sidebar-accent-foreground",
        "sidebar-border", "sidebar-ring",
    ];

    #[cfg(feature = "gallery")]
    #[marionette_macros::gallery_demo(key = "catalog-typography", name = "Catalog: Typography")]
    #[must_use]
    pub fn gallery_demo() -> Vec<Node> {
        // --- Card 1: Type scale ---
        let card1_heading = Heading::new("Type scale")
            .id("catalog-typo-card1-heading").level(2).build();
        let h1 = Heading::new("Heading 1 — sample").id("catalog-typo-h1").level(1).build();
        let h2 = Heading::new("Heading 2 — sample").id("catalog-typo-h2").level(2).build();
        let h3 = Heading::new("Heading 3 — sample").id("catalog-typo-h3").level(3).build();
        let h4 = Heading::new("Heading 4 — sample").id("catalog-typo-h4").level(4).build();
        let h5 = Heading::new("Heading 5 — sample").id("catalog-typo-h5").level(5).build();
        let h6 = Heading::new("Heading 6 — sample").id("catalog-typo-h6").level(6).build();
        let body = Text::new("Body text. The quick brown fox jumps over the lazy dog.")
            .id("catalog-typo-body").build();
        // Caption: wrap in Container because Text has no .class() prop (PATTERNS.md line 360-364)
        let caption_inner = Text::new("Caption / label — small descriptive text")
            .id("catalog-typo-caption-inner").build();
        let caption = Container::new()
            .id("catalog-typo-caption-wrapper")
            .class("text-xs text-muted-foreground")
            .children(vec![caption_inner.clone()])
            .build_with_children();

        let card1_kids = vec![
            card1_heading.clone(), h1, h2, h3, h4, h5, h6, body,
            caption.first().cloned().expect("caption root"),
        ];
        let card1 = Container::new().id("catalog-typo-card1")
            .class(CARD_CLASS).children(card1_kids).build_with_children();

        // --- Card 2: Lucide icon catalog (14 icons) ---
        let card2_heading = Heading::new("Lucide icon catalog")
            .id("catalog-typo-card2-heading").level(2).build();

        let mut icon_cells_roots: Vec<(String, marionette_protocol::Component)> =
            Vec::with_capacity(ICONS.len());
        let mut icon_descendants: Vec<Node> = Vec::new();

        for name in ICONS {
            let btn = Button::new("")
                .id(format!("catalog-typo-icon-{name}"))
                .variant("outline")
                .size("icon")
                .icon(name)
                .aria_label(format!("{name} icon"))
                .build();
            let label_inner = Text::new(name).id(format!("catalog-typo-icon-label-{name}")).build();
            let label_wrapper = Container::new()
                .id(format!("catalog-typo-icon-labelwrap-{name}"))
                .class("text-xs text-muted-foreground")
                .children(vec![label_inner.clone()])
                .build_with_children();
            let cell = Container::new()
                .id(format!("catalog-typo-icon-cell-{name}"))
                .class(ICON_CELL_CLASS)
                .children(vec![btn.clone(), label_wrapper.first().cloned().expect("labelwrap root")])
                .build_with_children();
            icon_cells_roots.push(cell[0].clone());
            icon_descendants.extend(cell.into_iter().skip(1));
            icon_descendants.extend(label_wrapper.into_iter().skip(1));
        }

        let icon_grid = Container::new()
            .id("catalog-typo-icon-grid")
            .class(ICON_GRID_CLASS)
            .children(icon_cells_roots.clone())
            .build_with_children();

        let card2 = Container::new().id("catalog-typo-card2").class(CARD_CLASS)
            .children(vec![card2_heading.clone(),
                icon_grid.first().cloned().expect("icon-grid root")])
            .build_with_children();

        // --- Card 3: OKLCH semantic tokens (26 colour + 1 radius) ---
        let card3_heading = Heading::new("OKLCH semantic tokens")
            .id("catalog-typo-card3-heading").level(2).build();

        let mut swatch_cells_roots: Vec<(String, marionette_protocol::Component)> =
            Vec::with_capacity(COLOUR_TOKENS.len() + 1);
        let mut swatch_descendants: Vec<Node> = Vec::new();

        for token in COLOUR_TOKENS {
            let colour_box = Container::new()
                .id(format!("catalog-typo-swatch-box-{token}"))
                .class(format!("w-full h-16 rounded-md border bg-{token}"))
                .build();
            let label_inner = Text::new(format!("--{token}"))
                .id(format!("catalog-typo-swatch-label-{token}")).build();
            let label_wrapper = Container::new()
                .id(format!("catalog-typo-swatch-labelwrap-{token}"))
                .class("text-xs font-mono text-muted-foreground")
                .children(vec![label_inner.clone()])
                .build_with_children();
            let cell = Container::new()
                .id(format!("catalog-typo-swatch-cell-{token}"))
                .class(SWATCH_CELL_CLASS)
                .children(vec![colour_box.clone(),
                    label_wrapper.first().cloned().expect("labelwrap root")])
                .build_with_children();
            swatch_cells_roots.push(cell[0].clone());
            swatch_descendants.extend(cell.into_iter().skip(1));
            swatch_descendants.extend(label_wrapper.into_iter().skip(1));
        }

        // Radius demo cell (UI-SPEC §CAT-05 line 631)
        let radius_box = Container::new()
            .id("catalog-typo-swatch-box-radius")
            .class("w-16 h-16 rounded border bg-muted")
            .build();
        let radius_label_inner = Text::new("--radius")
            .id("catalog-typo-swatch-label-radius").build();
        let radius_label_wrapper = Container::new()
            .id("catalog-typo-swatch-labelwrap-radius")
            .class("text-xs font-mono text-muted-foreground")
            .children(vec![radius_label_inner.clone()])
            .build_with_children();
        let radius_cell = Container::new()
            .id("catalog-typo-swatch-cell-radius")
            .class(SWATCH_CELL_CLASS)
            .children(vec![radius_box.clone(),
                radius_label_wrapper.first().cloned().expect("root")])
            .build_with_children();
        swatch_cells_roots.push(radius_cell[0].clone());
        swatch_descendants.extend(radius_cell.into_iter().skip(1));
        swatch_descendants.extend(radius_label_wrapper.into_iter().skip(1));

        let swatch_grid = Container::new()
            .id("catalog-typo-swatch-grid")
            .class(SWATCH_GRID_CLASS)
            .children(swatch_cells_roots.clone())
            .build_with_children();

        let card3 = Container::new().id("catalog-typo-card3").class(CARD_CLASS)
            .children(vec![card3_heading.clone(),
                swatch_grid.first().cloned().expect("swatch-grid root")])
            .build_with_children();

        // --- Title + outer assembly ---
        let title = Heading::new("Typography & Tokens")
            .id("catalog-typography-title").level(1).build();
        let intro = Text::new(
            "Type scale, lucide-svelte icon catalog, and OKLCH semantic colour tokens used by every screen in the gallery.",
        ).id("catalog-typography-intro").build();

        let outer = Container::new()
            .id("catalog-typography-root")
            .class(OUTER_CLASS)
            .children(vec![
                title.clone(), intro.clone(),
                card1.first().cloned().expect("card1 root"),
                card2.first().cloned().expect("card2 root"),
                card3.first().cloned().expect("card3 root"),
            ])
            .build_with_children();

        let mut result: Vec<Node> = Vec::new();
        result.extend(outer);
        // Card 1 descendants
        result.extend(card1.into_iter().skip(1));
        result.extend(caption.into_iter().skip(1));
        // Card 2 descendants
        result.extend(card2.into_iter().skip(1));
        result.extend(icon_grid.into_iter().skip(1));
        result.extend(icon_descendants);
        // Card 3 descendants
        result.extend(card3.into_iter().skip(1));
        result.extend(swatch_grid.into_iter().skip(1));
        result.extend(swatch_descendants);
        result
    }

    #[cfg(all(test, feature = "gallery"))]
    mod tests {
        use super::*;
        use marionette::gallery::registered_demos;

        #[test]
        fn root_id() {
            let v = gallery_demo();
            assert_eq!(v[0].0, "catalog-typography-root");
        }

        #[test]
        fn six_heading_levels_present() {
            let v = gallery_demo();
            for level in 1..=6 {
                let id = format!("catalog-typo-h{level}");
                let node = v.iter().find(|(i, _)| i == &id)
                    .unwrap_or_else(|| panic!("missing heading {id}"));
                let s = serde_json::to_value(&node.1).unwrap();
                assert_eq!(s["type"], "heading");
                assert_eq!(s["props"]["level"], level);
            }
        }

        #[test]
        fn fourteen_icon_buttons_with_locked_names() {
            let v = gallery_demo();
            for name in ICONS {
                let id = format!("catalog-typo-icon-{name}");
                let node = v.iter().find(|(i, _)| i == &id)
                    .unwrap_or_else(|| panic!("missing icon cell {id}"));
                let s = serde_json::to_value(&node.1).unwrap();
                assert_eq!(s["type"], "button");
                assert_eq!(s["props"]["variant"], "outline");
                assert_eq!(s["props"]["size"], "icon");
                assert_eq!(s["props"]["icon"], name);
                assert!(s["props"]["aria_label"].as_str().map_or(false, |s| !s.is_empty()));
            }
        }

        #[test]
        fn twenty_seven_swatch_cells() {
            let v = gallery_demo();
            let mut count = 0;
            for (id, _c) in &v {
                if id.starts_with("catalog-typo-swatch-cell-") {
                    count += 1;
                }
            }
            assert_eq!(count, 27, "expected 26 colour tokens + 1 radius = 27 swatch cells");
        }

        #[test]
        fn colour_swatch_boxes_use_bg_token_class() {
            let v = gallery_demo();
            for token in COLOUR_TOKENS {
                let id = format!("catalog-typo-swatch-box-{token}");
                let node = v.iter().find(|(i, _)| i == &id)
                    .unwrap_or_else(|| panic!("missing swatch box {id}"));
                let s = serde_json::to_value(&node.1).unwrap();
                let class = s["props"]["class"].as_str().unwrap_or("");
                let expected = format!("bg-{token}");
                assert!(class.contains(&expected),
                    "swatch {token} class missing {expected}: {class}");
            }
        }

        #[test]
        fn radius_swatch_cell_present() {
            let v = gallery_demo();
            let id = "catalog-typo-swatch-cell-radius";
            assert!(v.iter().any(|(i, _)| i == id), "missing radius cell");
        }

        #[test]
        fn registered_demos_includes_catalog_typography() {
            let e = registered_demos().find(|e| e.key == "catalog-typography").expect("registered");
            assert_eq!(e.name, "Catalog: Typography");
        }
    }
