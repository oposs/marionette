**Part A — Create `frontend/src/lib/components/form/Textarea.svelte`:**

```svelte
<script lang="ts">
	import * as Field from '$lib/components/ui/field';
	import { Textarea as ShadcnTextarea } from '$lib/components/ui/textarea';
	import { getData, setData } from '$lib/store/data.svelte';
	import { markDirty, clearDirty } from '$lib/store/dirty.svelte';
	import { sendAction } from '$lib/transport/dispatcher';
	import type { ComponentAction } from '$lib/transport/messages';

	let {
		props = {},
		bind,
		action,
		surface,
	}: {
		props: Record<string, unknown>;
		bind?: string;
		action?: ComponentAction;
		surface: string;
	} = $props();

	const fallbackId = crypto.randomUUID();
	let fieldId = $derived((props.id as string) ?? fallbackId);

	let value = $derived(bind ? ((getData(surface, bind) as string) ?? '') : '');
	let fieldError = $derived(
		bind ? ((getData(surface, '/_errors' + bind) as string) ?? '') : ''
	);
	let hasError = $derived(!!fieldError);

	function handleInput(e: Event) {
		if (bind) {
			const target = e.currentTarget as HTMLTextAreaElement;
			setData(surface, bind, target.value);
		}
	}

	function handleFocus() {
		if (bind) markDirty(bind);
	}

	function handleBlur() {
		if (bind) {
			clearDirty(bind, (op) => setData(surface, op.path, op.value));
			if (action?.type === 'blur') {
				sendAction(
					action.name ?? action.type,
					{ value: getData(surface, bind!) },
					action.target
				);
			}
		}
	}
</script>

<Field.Field
	data-invalid={hasError || undefined}
	class={props.full_width ? 'col-span-full' : undefined}
>
	{#if props.label}
		<Field.Label for={fieldId}>{props.label}</Field.Label>
	{/if}
	<ShadcnTextarea
		id={fieldId}
		placeholder={props.placeholder as string}
		rows={(props.rows as number) ?? 4}
		required={props.required as boolean}
		disabled={props.disabled as boolean}
		aria-invalid={hasError || undefined}
		{value}
		oninput={handleInput}
		onfocus={handleFocus}
		onblur={handleBlur}
	/>
	{#if props.description && !hasError}
		<Field.Description>{props.description}</Field.Description>
	{/if}
	{#if fieldError}
		<Field.Error>{fieldError}</Field.Error>
	{/if}
</Field.Field>
```

**Critical details:**

1. Same Field anatomy as TextInput (Plan 02). The ONLY structural differences: `ShadcnTextarea` instead of `Input`, `rows` prop forwarding (default 4 per 14-UI-SPEC.md §Textarea), `currentTarget` cast to `HTMLTextAreaElement` in `handleInput`.
2. `rows={(props.rows as number) ?? 4}` — Open Question OQ4 from RESEARCH: if the shadcn Textarea primitive does NOT forward `rows` to the underlying `<textarea>` (inspect `ui/textarea/textarea.svelte` — it almost certainly uses `{...restProps}` like Input does), add a `min-h-[{rows * 1.5}rem]` fallback via inline style. First test the direct-pass approach; fall back only if verified broken.
3. Same blur-race safety as TextInput — this component benefits transitively from Plan 01's NodeRenderer fix.
4. `data-invalid` and `aria-invalid` both use `|| undefined` (Pitfall #4).

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| User typed text → frontend → backend (blur/submit) | Textarea value reaches server via existing setData / ActionMessage paths |
| Handler → frontend (props) | label/description/placeholder strings flow into Field elements |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-14-05-01 | Tampering | Text injection via `props.label` / `props.description` / `props.placeholder` | mitigate | Svelte `{expression}` auto-escapes. Backend handlers control strings. |
| T-14-05-02 | Tampering | Text injection via `/_errors/{bind}` | mitigate | Svelte auto-escape; server authoritative for error text. |
| T-14-05-03 | Input Validation | User-pasted multiline input with embedded HTML/XSS payload | mitigate | Textarea value stored as a plain string in the data store; rendered via `{value}` which auto-escapes. Server-side validation (V5) still applies on submit. Client never evals pasted content. |
| T-14-05-04 | Denial of Service | Pasted megabyte-scale text causes reactivity storm | accept | Same risk class as any `<input>` in this app; server-side payload size limits (if configured) apply. Not addressed in Phase 14. |
| T-14-05-05 | Information Disclosure | Default `rows={4}` hides long pasted content requiring scroll | accept | UX concern, not security. `resize-y` default allows the user to expand. |
</threat_model>
