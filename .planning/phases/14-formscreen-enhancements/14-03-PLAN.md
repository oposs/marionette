<script lang="ts">
	import * as Field from '$lib/components/ui/field';
	import * as Select from '$lib/components/ui/select';
	import { getAllData, getData, setData } from '$lib/store/data.svelte';
	import { markDirty, clearDirty } from '$lib/store/dirty.svelte';
	import { sendAction } from '$lib/transport/dispatcher';
	import type { ComponentAction } from '$lib/transport/messages';

	let {
		props = {},
		bind,
		action,
		surface,
	}: {
		props: Record<string, unknown>;
		bind?: string;
		action?: ComponentAction;
		surface: string;
	} = $props();

	// D-B4: stable id — handler-supplied wins; mount-time UUID fallback.
	const fallbackId = crypto.randomUUID();
	let fieldId = $derived((props.id as string) ?? fallbackId);

	let value = $derived(bind ? ((getData(surface, bind) as string) ?? '') : '');
	let options = $derived(
		(props.options as Array<{ value: string; label: string }>) ?? []
	);
	let fieldError = $derived(
		bind ? ((getData(surface, '/_errors' + bind) as string) ?? '') : ''
	);
	let hasError = $derived(!!fieldError);

	function handleValueChange(newValue: string) {
		if (bind) {
			setData(surface, bind, newValue);
		}
		// Phase 12 node-patch demo — D-A6 focus preservation.
		// Keep verbatim: surface data + action.payload merge.
		if (action?.type === 'change' && action.name) {
			const surfaceData = getAllData(surface) ?? {};

				...((action.payload as Record<string, unknown>) ?? {}),
				...surfaceData,
			};
			sendAction(action.name, payload, action.target);
		}
	}

	function handleOpenChange(open: boolean) {
		if (!bind) return;
		if (open) {
			markDirty(bind);
		} else {
			clearDirty(bind, (op) => setData(surface, op.path, op.value));
		}
	}
</script>

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Handler → frontend (props payload) | Backend-controlled option labels, description, error text flow into Field.Label / Field.Description / Field.Error / Select.Item |
| User selection → frontend → backend (change action) | Selected value reaches server via `ActionMessage` with merged surface data |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-14-03-01 | Tampering | Text injection via `props.label` / `props.description` / `options[].label` into Field.Label / Field.Description / Select.Item | mitigate | Svelte `{expression}` auto-escapes. No `{@html}` used. Backend handlers are sole source of these strings. |
| T-14-03-02 | Tampering | Error text injection via `/_errors/{bind}` | mitigate | Svelte auto-escape. Server is authoritative for error content. Unchanged from Phase 11. |
| T-14-03-03 | Spoofing | Change-action dispatch payload includes unsanitized surface data | mitigate | Payload merge is `{ ...action.payload, ...surfaceData }` — same as Phase 12; surface data is server-seeded and user-input-sanitized at input time. Handlers validate on the backend. Unchanged from Phase 12. |
| T-14-03-04 | Information Disclosure | Full surface data leaks in change-action payload | accept | Intentional — Phase 12 country-select demo requires the full contact form state to be sent on country change so the server can re-validate cross-field rules. Server-side authz gates who receives the payload. |
| T-14-03-05 | Tampering | Label/trigger id collision between two SelectInputs on the same page | mitigate | UUID fallback for unprompted ids; handler-supplied ids are the norm in CRM (e.g., `contact-form-company`) — collisions are a handler bug, not a component concern. |
</threat_model>
