    export interface PerfSnapshot {
        ttfp_ms: number | null;
        fps: number | null;
        memory_mb: number | null;
        latency_p95_ms: number | null;
    }

    // ---- TTFP — buffered read (Pitfall 3) ----
    export function captureTTFP(): number | null {
        const entries = performance.getEntriesByType('paint');
        const fp = entries.find((e) => e.name === 'first-paint');
        return fp ? fp.startTime : null;
    }

    // ---- Scroll FPS — rAF delta over 5 s window ----
    const FPS_SAMPLE_WINDOW_MS = 5_000;

    export function startFpsSampler(onDone: (fps: number) => void): () => void {
        const samples: number[] = [];
        let last = performance.now();
        let running = true;
        const startTime = last;

        const loop = (t: number) => {
            if (!running) return;
            const dt = t - last;
            if (dt > 0) samples.push(1000 / dt);
            last = t;
            if (t - startTime > FPS_SAMPLE_WINDOW_MS) {
                samples.sort((a, b) => a - b);
                const fps = samples[Math.floor(samples.length / 2)] ?? 0;
                onDone(fps);
                running = false;
                return;
            }
            requestAnimationFrame(loop);
        };
        requestAnimationFrame(loop);
        return () => { running = false; };
    }

    // ---- Memory — Chromium-only guard (Pitfall 4) ----
    export function captureMemoryMb(): number | null {
        const p = performance as Performance & { memory?: { usedJSHeapSize: number } };
        if (!p.memory || typeof p.memory.usedJSHeapSize !== 'number') return null;
        return p.memory.usedJSHeapSize / (1024 * 1024);
    }

    // ---- Patch-apply latency — via init.ts instrumentation hook ----
    const LATENCY_BUFFER_SIZE = 100;
    const latencyBuffer: number[] = [];

    export function recordPatchLatency(ms: number): void {
        if (latencyBuffer.length >= LATENCY_BUFFER_SIZE) latencyBuffer.shift();
        latencyBuffer.push(ms);
    }

    export function getLatencyP95(): number | null {
        if (latencyBuffer.length === 0) return null;
        const sorted = [...latencyBuffer].sort((a, b) => a - b);
        const idx = Math.floor(sorted.length * 0.95);
        return sorted[Math.min(idx, sorted.length - 1)];
    }

    // ---- Round-trip report ----
    export function reportPerf(snapshot: PerfSnapshot): void {
        sendAction('gallery-demo/exer-03/report-perf', snapshot);
    }

    // ---- Auto-arm on EXER-03 screen mount ----
    //
    // Detect the EXER-03 perf readout cell in the DOM and wire everything:
    // 1. Install patch-latency probe (records into latencyBuffer).
    // 2. Capture TTFP + memory(t0) on first detection; memory(t30) after 30 s.
    // 3. Listen for first scroll on the DataTable → start FPS sampler.
    // 4. Report snapshot on mount + after 30 s + on Remeasure-tick data patch.

    if (typeof window !== 'undefined') {
        let armed = false;
        let memoryT0: number | null = null;

        const arm = () => {
            if (armed) return;
            if (!document.getElementById('exer-03-perf-ttfp')) return;
            armed = true;

            // Install the latency probe immediately (before ticks flow in).
            installPatchProbe((dt) => recordPatchLatency(dt));

            // Capture initial values after a microtask so the paint entry is buffered.
            setTimeout(() => {
                memoryT0 = captureMemoryMb();
                const snapshot: PerfSnapshot = {
                    ttfp_ms: captureTTFP(),
                    fps: null,
                    memory_mb: null,          // growth captured at t30, not t0
                    latency_p95_ms: getLatencyP95(),
                };
                reportPerf(snapshot);
            }, 100);

            // At t+30s, report memory GROWTH + refreshed latency.
            setTimeout(() => {
                const m = captureMemoryMb();
                const growth = (m !== null && memoryT0 !== null) ? m - memoryT0 : null;
                reportPerf({
                    ttfp_ms: null,
                    fps: null,
                    memory_mb: growth,
                    latency_p95_ms: getLatencyP95(),
                });
            }, 30_000);

            // First scroll → kick off FPS sampler. Listen on window (capture phase)
            // so we catch the DataTable scroll container too.
            const scrollHandler = () => {
                window.removeEventListener('scroll', scrollHandler, true);
                startFpsSampler((fps) => {
                    reportPerf({ ttfp_ms: null, fps, memory_mb: null, latency_p95_ms: null });
                });
            };
            window.addEventListener('scroll', scrollHandler, true);
        };

        const obs = new MutationObserver(() => arm());
        obs.observe(document.body, { childList: true, subtree: true });
        if (document.readyState !== 'loading') arm();
        else document.addEventListener('DOMContentLoaded', arm);

        // Wire Remeasure button: when data-store patches /demo/exer-03/perf/remeasure-tick,
        // re-capture everything. Simpler implementation: wire a click listener on
        // the #exer-03-remeasure button.
        document.addEventListener('click', (ev) => {
            const t = ev.target as HTMLElement | null;
            if (!t?.closest('#exer-03-remeasure')) return;
            // Fire a fresh full snapshot (TTFP only returns the original first-paint;
            // memory + latency are live).
            const m = captureMemoryMb();
            const growth = (m !== null && memoryT0 !== null) ? m - memoryT0 : null;
            reportPerf({
                ttfp_ms: captureTTFP(),
                fps: null,        // FPS requires a scroll window; user must scroll to retrigger
                memory_mb: growth,
                latency_p95_ms: getLatencyP95(),
            });
        }, { passive: true });
    }
