   Session handler using tokio::sync::mpsc channel pattern (per research recommendation):
   ```
   async fn handle_session(socket: WebSocket, state: Arc<AppState>) {
       let (ws_sender, ws_receiver) = socket.split();
       let (tx, rx) = tokio::sync::mpsc::channel::<ProtocolMessage>(32);
       let session = WsSession::new();

       // Writer task: drains mpsc channel and sends to WS
       let write_task = tokio::spawn(write_loop(ws_sender, rx));

       // Send hello message
       let _ = tx.send(ProtocolMessage::Hello(HelloMessage { version: "1.0.0".into() })).await;

       // Reader task: reads WS messages, dispatches actions, sends responses through tx
       read_loop(ws_receiver, tx, &state, &session).await;

       // Clean up
       write_task.abort();
   }
   ```

   read_loop: Process incoming messages:
   - Message::Text(text) -> parse as ActionMessage via serde_json::from_str
   - If parse fails, send ErrorMessage through tx
   - If parse succeeds, build HandlerContext and call state.router.dispatch()
   - Send each response ProtocolMessage through tx
   - Message::Ping(data) -> send Pong(data) (axum may handle this automatically)
   - Message::Close(_) -> break loop

   write_loop: Drain mpsc receiver and send as Message::Text(serde_json::to_string(&msg)):
   - while let Some(msg) = rx.recv().await { sender.send(Message::Text(...)).await }
