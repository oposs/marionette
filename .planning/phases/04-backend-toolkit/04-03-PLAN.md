1. Create backend/crates/marionette/src/auth.rs:
   ```
   pub fn check_auth(requirement: &AuthRequirement, session: &Session) -> Result<(), ActionError> {
       match requirement {
           AuthRequirement::None => Ok(()),
           AuthRequirement::Authenticated => {
               if session.user_id.is_some() { Ok(()) }
               else { Err(ActionError::Unauthorized("Authentication required".into())) }
           }
           AuthRequirement::Role(role) => {
               if session.user_id.is_none() {
                   return Err(ActionError::Unauthorized("Authentication required".into()));
               }
               if session.roles.iter().any(|r| r == role) { Ok(()) }
               else { Err(ActionError::Unauthorized(format!("Role '{}' required", role))) }
           }
       }
   }
   ```
