```ts
import { describe, it, expect, beforeEach, afterEach } from 'vitest';
import { render } from 'vitest-browser-svelte';
import NodeRenderer from './NodeRenderer.svelte';
import type { ComponentNode } from '$lib/transport/messages';

describe('NodeRenderer unmount race (D-E2)', () => {
    let capturedErrors: ErrorEvent[] = [];
    const errorHandler = (e: ErrorEvent) => capturedErrors.push(e);

    beforeEach(() => {
        capturedErrors = [];
        window.addEventListener('error', errorHandler);
    });
    afterEach(() => {
        window.removeEventListener('error', errorHandler);
    });

    it('does not throw TypeError when a bound node is removed while rendered', async () => {
        const nodes: Record<string, ComponentNode> = $state({
            root: { type: 'text-input', bind: '/x', children: [] }
        });
        const { rerender } = render(NodeRenderer, {
            nodeId: 'root',
            nodes,
            surface: 'test'
        });
        // Trigger the race: remove `root` from nodes and rerender.
        delete nodes.root;
        await rerender({ nodeId: 'root', nodes, surface: 'test' });
        await new Promise(r => setTimeout(r, 50));
        const typeErrors = capturedErrors.filter(e =>
            /Cannot read properties of undefined \(reading 'bind'\)/.test(e.message)
        );
        expect(typeErrors.length).toBe(0);
    });
});
```
