   const registry = new Map<string, Component>();
   export function register(type: string, component: Component): void { registry.set(type, component); }
   export function getComponent(type: string): Component | undefined { return registry.get(type); }
   export function registerAll(components: Record<string, Component>): void {

6. Create `frontend/src/lib/components/core/NodeRenderer.svelte`:
   ```svelte
   <script lang="ts">
     import { getComponent } from '$lib/registry/registry';
     import { getData } from '$lib/store/data.svelte';
     import FallbackComponent from './FallbackComponent.svelte';
     import ErrorBoundary from './ErrorBoundary.svelte';
     import type { ComponentNode } from '$lib/transport/messages';

     let { nodeId, nodes, surface }: {
       nodeId: string;
       nodes: Record<string, ComponentNode>;
       surface: string;
     } = $props();

     let node = $derived(nodes[nodeId]);
     let Component = $derived(node ? (getComponent(node.type) ?? FallbackComponent) : null);
   </script>

   {#if node}
     {#if !node.visible || getData(surface, node.visible)}
       <ErrorBoundary>
         {#if Component === FallbackComponent}
           <FallbackComponent nodeType={node.type} props={node.props} {surface} />
         {:else if Component}
           <Component
             props={node.props ?? {}}
             bind={node.bind}
             action={node.action}
             {surface}
           >
             {#snippet children()}
               {#if node.children}
                 {#each node.children as childId (childId)}
                   <svelte:self nodeId={childId} {nodes} {surface} />
                 {/each}
               {/if}
             {/snippet}
           </Component>
         {/if}
       </ErrorBoundary>
     {/if}
   {/if}
   ```
   Note: Use the Svelte MCP server to verify correct Svelte 5 syntax for dynamic components, snippets, and self-references. Adjust the pattern based on what Svelte 5 actually supports for passing children to dynamic components.
