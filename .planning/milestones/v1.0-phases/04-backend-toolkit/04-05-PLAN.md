#[tokio::test]
async fn test_create_session() {
    let db = marionette::test_db().await;
    let session = marionette::session::ActiveModel {
        session_token: Set("tok-abc123".into()),
        session_roles: Set("[]".into()),
        session_created: Set("2026-01-01T00:00:00".into()),

    assert!(result.session_id > 0);
    assert_eq!(result.session_token, "tok-abc123");
}
